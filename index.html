import re def password_strength(password): checks = { "بەڵگەی درێژی": len(password) >= 12, "پیت بچووک": bool(re.search(r"[a-z]", password)), "پیت گەورە": bool(re.search(r"[A-Z]", password)), "ژمارە": bool(re.search(r"\d", password)), "هێمای تایبەت": bool(re.search(r"[^A-Za-z0-9]", password)), } passed = sum(checks.values()) for name, ok in checks.items(): print(f"{name}: {'باشە ✓' if ok else 'پێویستی بە باشترکردنە'}") if passed == len(checks): print("\nئەنجام: پاسۆردەکەت بەهێزە.") else: print("\nئەنجام: پاسۆردێکی درێژتر و جیاوازتر بەکاربهێنە.") password = input("پاسۆردی خۆت بنووسە: ") password_strength(password)
