/* ========================================== DOMINO GAME ENGINE (PRO SYSTEM) ========================================== */ class DominoGame { constructor() { this.deck = []; this.playerHand = []; this.botHand = []; this.board = []; // زنجیرەی دۆمینە دانراوەکانی سەر مێزەکە this.boneyard = []; // بەردە ماوەکان (سندوق) this.turn = "player"; // 'player' یان 'bot' } // 1. دروستکردنی 28 بەردەکەی دۆمینە (لە 0-0 تا 6-6) generateDeck() { this.deck = []; for (let i = 0; i <= 6; i++) { for (let j = i; j <= 6; j++) { this.deck.push({ left: i, right: j, id: `${i}-${j}` }); } } } // 2. تێکدانی بەردەکان (Shuffle) shuffleDeck() { for (let i = this.deck.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [this.deck[i], this.deck[j]] = [this.deck[j], this.deck[i]]; } } // 3. دابەشکردنی بەردەکان (7 بەرد بۆ هەر یاریزانێک) deal() { this.generateDeck(); this.shuffleDeck(); this.playerHand = this.deck.splice(0, 7); this.botHand = this.deck.splice(0, 7); this.boneyard = [...this.deck]; this.deck = []; } // 4. دۆزینەوەی سەرەتای یاری (ئەو یاریزانەی جفتە شێخی 6-6ی پێیە دەستپێدەکات) determineFirstTurn() { const hasDoubleSix = (hand) => hand.some(tile => tile.left === 6 && tile.right === 6); if (hasDoubleSix(this.playerHand)) { this.turn = "player"; } else if (hasDoubleSix(this.botHand)) { this.turn = "bot"; } else { this.turn = "player"; // ئەگەر لای هیچکامیان نەبوو } } // 5. پشکنینی ڕاستیی جوڵەکە (Valid Move Validation) canPlayTile(tile) { if (this.board.length === 0) return true; // یەکەم بەردی سەر مێزەکە const leftEnd = this.board[0].left; const rightEnd = this.board[this.board.length - 1].right; return ( tile.left === leftEnd || tile.right === leftEnd || tile.left === rightEnd || tile.right === rightEnd ); } // 6. دانانی بەرد لەسەر مێزەکە (سەر یان خوارەوەی زنجیرەکە) playTile(playerType, tileIndex, side = "right") { const hand = playerType === "player" ? this.playerHand : this.botHand; const tile = hand[tileIndex]; if (!this.canPlayTile(tile)) { return { success: false, message: "جوڵەکە ڕێگەپێدراو نییە!" }; } // لۆجیکی یەکەم بەرد if (this.board.length === 0) { this.board.push(tile); } else { const leftEnd = this.board[0].left; const rightEnd = this.board[this.board.length - 1].right; if (side === "left") { if (tile.right === leftEnd) { this.board.unshift(tile); } else if (tile.left === leftEnd) { this.board.unshift({ left: tile.right, right: tile.left, id: tile.id }); } } else { // side === 'right' if (tile.left === rightEnd) { this.board.push(tile); } else if (tile.right === rightEnd) { this.board.push({ left: tile.right, right: tile.left, id: tile.id }); } } } // لابردنی بەردەکە لە دەستی یاریزانەکە hand.splice(tileIndex, 1); // گوێستنەوەی سەرپۆرەی یاریزان this.turn = this.turn === "player" ? "bot" : "player"; return { success: true, board: this.board }; } // 7. ڕاکێشانی بەرد لە سندوق (Boneyard Pull) drawTile(playerType) { if (this.boneyard.length === 0) return null; const tile = this.boneyard.pop(); const hand = playerType === "player" ? this.playerHand : this.botHand; hand.push(tile); return tile; } // 8. پشکنینی براوەی یاری (Check Winner) checkWin() { if (this.playerHand.length === 0) return "یاریزان براوە بوو! 🎉"; if (this.botHand.length === 0) return "کۆمپیوتەر براوە بوو! 🤖"; return null; } }
